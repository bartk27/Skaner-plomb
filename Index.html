// AUTOMATYCZNY ZAPIS DANYCH function saveBackup(){ let data={ registration: document.getElementById("registration").value, operator: document.getElementById("operator").value, scans:scans, unloadSeconds:unloadSeconds, cleanSeconds:cleanSeconds }; localStorage.setItem( "currentReport", JSON.stringify(data) ); } setInterval(saveBackup,5000); // WCZYTANIE PO URUCHOMIENIU window.onload=function(){ let saved= localStorage.getItem("currentReport"); if(saved){ let data=JSON.parse(saved); document.getElementById("registration").value= data.registration || ""; document.getElementById("operator").value= data.operator || ""; scans=data.scans || []; unloadSeconds=data.unloadSeconds || 0; cleanSeconds=data.cleanSeconds || 0; update(); } showHistory(); }; // EKSPORT EXCEL function exportExcel(){ let reg= document.getElementById("registration").value || "BRAK"; let operator= document.getElementById("operator").value || "BRAK"; let rows=[]; rows.push([ "Numer", "Plomba", "Typ", "Godzina" ]); scans.forEach((x,i)=>{ rows.push([ i+1, x, x.substring(0,2), new Date().toLocaleString() ]); }); rows.push([]); rows.push([ "Rejestracja", reg ]); rows.push([ "Operator", operator ]); rows.push([ "Plomb 01", scans.filter(x=>x.startsWith("01")).length ]); rows.push([ "Plomb 05", scans.filter(x=>x.startsWith("05")).length ]); rows.push([ "Razem", scans.length ]); rows.push([ "Wydajność plomb/min", (scans.length/unloadSeconds*60).toFixed(2) ]); let csv="\ufeff"; rows.forEach(r=>{ csv+=r.join(";")+"\n"; }); let blob=new Blob( [csv], { type: "application/vnd.ms-excel" } ); let a=document.createElement("a"); a.href= URL.createObjectURL(blob); a.download= new Date().toISOString().slice(0,10) +"_"+reg+".xlsx"; a.click(); saveHistory(); } // ZAPIS HISTORII function saveHistory(){ let history= JSON.parse( localStorage.getItem("history") || "[]" ); history.push({ date: new Date().toLocaleString(), registration: document.getElementById("registration").value, operator: document.getElementById("operator").value, count: scans.length, data:scans }); localStorage.setItem( "history", JSON.stringify(history) ); showHistory(); } // WYŚWIETLENIE HISTORII function showHistory(){ let box= document.getElementById("history"); if(!box)return; let history= JSON.parse( localStorage.getItem("history") || "[]" ); box.innerHTML=""; history.reverse().forEach((h,i)=>{ box.innerHTML+=`
${h.date}
Auto: ${h.registration}
Operator: ${h.operator}
Plomby: ${h.count}
`; }); } // NOWY ROZŁADUNEK function newReport(){ if(confirm( "Zapisać obecny raport i rozpocząć nowy?" )){ saveHistory(); scans=[]; unloadSeconds=0; cleanSeconds=0; startUnloadTime=null; startCleanTime=null; document.getElementById("registration").value=""; document.getElementById("operator").value=""; update(); } }
